// ===== A2. Lista de la compra: function anadir() =====

// En este paso con function añado a la lista el producto que escribo, busco la caja y guardo el texto que contiene con el .value
// después he creado una etiqueta con li para el nuevo producto q añada, he creado un nodo de texto con textNode
//meto todo lo nuevo que he escrito en la lista con appenChild y la he guardado en una variable con var
//por último 
function anadir() {
 
    var producto = document.getElementById("producto").value;
    var elemento = document.createElement("li");
    var texto = document.createTextNode(producto);
    elemento.appendChild(texto);
    var lista = document.getElementById("lista");
    lista.appendChild(elemento);
}
